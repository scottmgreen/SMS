1. **Implement Distributed Caching**:
   ```csharp
   public class CachedHazardService : IHazardDataService
   {
       private readonly IMemoryCache _cache;
       private readonly IHazardDataService _innerService;
       
       public async Task<List<Hazard>> GetAllAsync(CancellationToken cancellationToken)
       {
           return await _cache.GetOrCreateAsync("all-hazards", async entry =>
   ```
