🌐 Step 3: IIS Configuration
1. Create Website Directory
# Create application directory
New-Item -Path "C:\inetpub\wwwroot\SMS3" -ItemType Directory -Force

# Copy published files (Mounted Iso Drive as source)
Copy-Item -Path "*MountedDrive*C:\SMS3Deploy\*" -Destination "D:\inetpub\wwwroot\SMS3" -Recurse -Force

2. Set Permissions
# Grant IIS_IUSRS permissions
icacls "C:\inetpub\wwwroot\SMS3" /grant "IIS_IUSRS:(OI)(CI)F" /T
icacls "C:\inetpub\wwwroot\SMS3" /grant "IIS AppPool\SMS3AppPool:(OI)(CI)F" /T

3. Create Application Pool
# Import WebAdministration module
Import-Module WebAdministration

# Create new application pool
New-WebAppPool -Name "SMS3AppPool"
Set-ItemProperty -Path "IIS:\AppPools\SMS3AppPool" -Name "managedRuntimeVersion" -Value ""
Set-ItemProperty -Path "IIS:\AppPools\SMS3AppPool" -Name "enable32BitAppOnWin64" -Value $false
Set-ItemProperty -Path "IIS:\AppPools\SMS3AppPool" -Name "processModel.identityType" -Value "ApplicationPoolIdentity"

4. Create IIS Website
# Remove default website (optional)
Remove-Website -Name "Default Web Site" -ErrorAction SilentlyContinue

# Create new website
New-Website -Name "SMS3" -Port 80 -PhysicalPath "D:\inetpub\wwwroot\SMS3" -ApplicationPool "SMS3AppPool"

🔧 Step 4: Configure web.config
<?xml version="1.0" encoding="utf-8"?>
<configuration>
  <location path="." inheritInChildApplications="false">
    <system.webServer>
      <handlers>
        <add name="aspNetCore" path="*" verb="*" modules="AspNetCoreModuleV2" resourceType="Unspecified" />
      </handlers>
      <aspNetCore processPath="dotnet" arguments=".\SMS3.dll" 
                  stdoutLogEnabled="true" 
                  stdoutLogFile=".\logs\stdout" 
                  hostingModel="inprocess">
        <environmentVariables>
          <environmentVariable name="ASPNETCORE_ENVIRONMENT" value="Production" />
        </environmentVariables>
      </aspNetCore>
    </system.webServer>
  </location>
</configuration>
